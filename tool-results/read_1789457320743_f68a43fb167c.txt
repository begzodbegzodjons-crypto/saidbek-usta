   645→
   646→function SummaryCard({
   647→  title,
   648→  value,
   649→  hint,
   650→  suffix,
   651→  tone = 'default',
   652→}: {
   653→  title: string
   654→  value: string
   655→  hint?: string
   656→  suffix?: string
   657→  tone?: 'default' | 'green' | 'red' | 'amber'
   658→}) {
   659→  const toneCls =
   660→    tone === 'green'
   661→      ? 'text-emerald-700'
   662→      : tone === 'red'
   663→        ? 'text-rose-700'
   664→        : tone === 'amber'
   665→          ? 'text-amber-700'
   666→          : ''
   667→  return (
   668→    <Card className="p-4">
   669→      <p className="text-xs font-medium text-muted-foreground">{title}</p>
   670→      <p className={`mt-1 text-lg font-bold tabular-nums lg:text-xl ${toneCls}`}>
   671→        {value}
   672→        {suffix && <span className="ml-1 text-xs font-normal text-muted-foreground">{suffix}</span>}
   673→      </p>
   674→      {hint && <Badge variant="secondary" className="mt-1.5">{hint}</Badge>}
   675→    </Card>
   676→  )
   677→}
